/**
 * @file polydim_solver_abi.h
 * @brief Canonical C ABI for POLYDIM V808 Architecture:
 *        - Stiefel Manifold Optimization St(D, K)
 *        - Banked Double-Buffer RCU Slot Leases (Zero-Copy IPC PMTP)
 *        - SPSC Wait-Free Telemetry Ring Buffer (128B Cache-Line Isolated)
 *        - Strict Allocator Pairing & Protected Refcounting
 * @copyright POLYDIM Architecture - 2026
 */

#ifndef POLYDIM_SOLVER_ABI_H
#define POLYDIM_SOLVER_ABI_H

#include <stdint.h>
#include <stddef.h>

#ifdef __cplusplus
#include <atomic>
extern "C" {
#endif

/* Terminal Status Codes */
typedef enum {
    POLYDIM_STATUS_OK                  =  0,
    POLYDIM_STATUS_CONVERGED_GRADIENT  =  1,
    POLYDIM_STATUS_CONVERGED_STEP      =  2,
    POLYDIM_STATUS_MAX_ITERATIONS      =  3,
    POLYDIM_STATUS_ERR_NULL_PTR        = -1,
    POLYDIM_STATUS_ERR_INVALID_DIM     = -2,
    POLYDIM_STATUS_ERR_NUMERICAL_NAN   = -3,
    POLYDIM_STATUS_ERR_DEGENERATE_NORM = -4,
    POLYDIM_STATUS_ERR_ORTHO_VIOLATION = -5,
    POLYDIM_STATUS_ERR_ALLOC           = -6,
    POLYDIM_STATUS_ERR_RING_FULL       = -7,
    POLYDIM_STATUS_ERR_RING_EMPTY      = -8,
    POLYDIM_STATUS_ERR_RANK_DEFICIENT  = -9,
    POLYDIM_STATUS_ERR_WRITER_BUSY     = -10,
    POLYDIM_STATUS_ERR_NO_FREE_SLOT    = -11,
    POLYDIM_STATUS_ERR_UNKNOWN         = -99
} PolydimStatusCode;

/* Stiefel Retraction Types */
typedef enum {
    POLYDIM_RETRACTION_CHOLQR2        = 0,
    POLYDIM_RETRACTION_CAYLEY_SMW     = 1,
    POLYDIM_RETRACTION_QR_HOUSEHOLDER = 2,
    POLYDIM_RETRACTION_SHIFTED_CHOLQR = 3
} PolydimRetractionType;

#define POLYDIM_CACHE_LINE 128
#define POLYDIM_ALIGN_128 alignas(POLYDIM_CACHE_LINE)
#define PMTP_MAX_READERS_PER_BANK 16

/* Optimizer Options Configuration */
#pragma pack(push, 8)
typedef struct {
    uint64_t max_iterations;
    double   gradient_tolerance;
    double   step_tolerance;
    double   objective_tolerance;
    double   ortho_tolerance;       /* Orthogonality tolerance ||X^T X - I||_F */
    uint32_t retraction_type;       /* PolydimRetractionType */
    uint32_t sampling_period;       /* Telemetry sample period */
    uint32_t num_threads;           /* OpenMP threads */
    double   learning_rate;         /* Step size alpha */
    double   shift_regularization;  /* Adaptive shift regularizer */
} PolydimSolverOptions;

/* Telemetry Sampling Point */
typedef struct {
    uint64_t iteration;
    double   objective_value;
    double   gradient_norm;
    double   step_size;
    double   ortho_error;          /* ||X^T X - I||_F */
    uint64_t elapsed_time_ns;
} PolydimTelemetryPoint;

/* Preallocated Telemetry Buffer */
typedef struct {
    PolydimTelemetryPoint* points;
    size_t   capacity;
    size_t   recorded_count;
} PolydimTelemetryBuffer;

/* High-Frequency Telemetry Event (Exactly 64 bytes) */
typedef struct {
    uint64_t timestamp_ns;
    uint32_t thread_id;
    uint32_t event_type;
    uint64_t iteration;
    double   objective_value;
    double   gradient_norm;
    double   ortho_error;
    double   step_size;
    uint64_t reserved;              /* 64-byte alignment padding */
} PolydimTelemetryEvent;

/* SPSC Wait-Free Ring Buffer (Isolated 128B Cache-Lines) */
typedef struct {
    alignas(128) uint64_t write_index;
    alignas(128) uint64_t read_index;
    alignas(128) uint64_t capacity;       /* Power of 2 */
    uint64_t capacity_mask;
    PolydimTelemetryEvent* ring_buffer;
} PolydimSpscRing;

/* Strict Allocator Pairing & Protected Refcounted Handle */
typedef struct {
    void*    data;
    size_t   bytes;
    int32_t  refcount;
    uint32_t flags;
    uint64_t allocation_id;
} PolydimHandle;

/* Solver Optimization Result */
typedef struct {
    int32_t  status;                /* PolydimStatusCode */
    uint64_t iterations_executed;
    double   final_objective;
    double   final_grad_norm;
    double   final_ortho_error;
    uint64_t total_time_ns;
    char     status_message[256];
} PolydimSolverResult;

typedef enum {
    PMTP_LEASE_FREE      = 0,
    PMTP_LEASE_ACTIVE    = 1,
    PMTP_LEASE_RECLAIMED = 2,
    PMTP_LEASE_CLOSED    = 3
} PmtpLeaseState;

/* Individual Reader Lease (Exactly 32 bytes, 8-byte aligned) */
typedef struct {
    uint32_t pid;
    uint32_t state;                 /* PmtpLeaseState */
    uint64_t process_start_time_ns;
    uint64_t generation;
    uint64_t acquired_ns;
} PmtpReaderLease;

/* Banked Double-Buffer Slot Header (Exactly 128B Header + 2x512B Leases = 1152B) */
typedef struct POLYDIM_ALIGN_128 {
    uint32_t active_bank;           /* 0 or 1 */
    uint32_t writer_active;         /* 1 if writer active */
    uint64_t sequence;              /* Monotonic generation */
    uint32_t owner_pid;             /* Writer process ID */
    uint32_t pad0;                  /* Alignment pad for 64-bit field */
    uint64_t owner_start_time_ns;   /* Writer timestamp */
    uint32_t num_reclaimed_orphans; /* Reclaimed counter */
    uint32_t pad1;                  /* Alignment pad */
    uint8_t  header_padding[88];    /* Header padding to exactly 128 bytes */
    PmtpReaderLease leases_bank0[PMTP_MAX_READERS_PER_BANK];
    PmtpReaderLease leases_bank1[PMTP_MAX_READERS_PER_BANK];
} PmtpBankedSlotHeader;
#pragma pack(pop)

/* Static ABI Verification */
#ifdef __cplusplus
static_assert(sizeof(PolydimTelemetryEvent) == 64, "PolydimTelemetryEvent must be exactly 64 bytes");
static_assert(sizeof(PmtpReaderLease) == 32, "PmtpReaderLease must be exactly 32 bytes");
static_assert(offsetof(PmtpBankedSlotHeader, leases_bank0) == 128, "leases_bank0 must start at 128B boundary");
static_assert(offsetof(PmtpBankedSlotHeader, leases_bank1) == 128 + 512, "leases_bank1 must start at 640B boundary");
static_assert(sizeof(PmtpBankedSlotHeader) == 1152, "PmtpBankedSlotHeader must be exactly 1152 bytes");
#endif
